```dart
Future<void> _runAnalyze(ArgResults command) async {
  final path = command['path'] as String;
  print('Analyzing project at: $path\n');

  final migrator = ShardI18nMigrator(verbose: true);
  final analysis = await migrator.analyze(path);

  print('Analysis Results:');
  print('${'─' * 60}');
  print('Total string literals found: ${analysis.totalStrings}');
  print('Extractable UI strings: ${analysis.extractableStrings}');
  print('Technical/code strings: ${analysis.technicalStrings}');
  print('Ambiguous strings (require review): ${analysis.ambiguousStrings}');
  print('');
  print('Interpolation patterns detected: ${analysis.interpolationCount}');
  print('Plural patterns detected: ${analysis.pluralCount}');
  print('');
  print('Average confidence score: ${analysis.averageConfidence.toStringAsFixed(1)}%');
  print('${'─' * 60}');
}

Future<void> _runMigrate(ArgResults command) async {
  final path = command['path'] as String;
  final dryRun = command['dry-run'] as bool;
  final auto = command['auto'] as bool;
  final configPath = command['config'] as String;

  print('${dryRun ? "Previewing" : "Starting"} migration for: $path');
  print('Mode: ${auto ? "Automatic" : "Interactive"}');
  if (dryRun) print('DRY RUN - No files will be modified');
  print('');

  final migrator = ShardI18nMigrator(verbose: true);

  MigrationConfig? config;
  if (File(configPath).existsSync()) {
    config = await MigrationConfig.load(configPath);
    print('Loaded configuration from: $configPath\n');
  }

  final result = await migrator.migrate(
    path,
    config: config,
    dryRun: dryRun,
    interactive: !auto,
  );

  print('\nMigration ${dryRun ? "Preview" : "Complete"}!');
  print('${'─' * 60}');
  print('Files created:');
  for (final file in result.createdFiles) {
    print('  + $file');
  }
  print('Files modified:');
  for (final file in result.modifiedFiles) {
    print('  ~ $file');
  }
  print('Strings extracted: ${result.stringsExtracted}');
  print('JSON keys generated: ${result.jsonKeysGenerated}');
  print('${'─' * 60}');
}

Future<void> _runInit() async {
  const configPath = 'migration_config.yaml';
  if (File(configPath).existsSync()) {
    print('Configuration file already exists: $configPath');
    print('Delete it first if you want to regenerate.');
    exit(1);
  }
  final config = MigrationConfig.createDefault();
  await config.save(configPath);
  print('Created configuration file: $configPath');
  print('Edit this file to customize the migration process.');
}
```

import 'dart:convert';
import 'dart:io';

import 'package:path/path.dart' as p;
import 'package:test/test.dart';

// We test the helper functions directly by importing them
// from the CLI bin file. Since they're top-level functions,
// we replicate the logic here for unit testing.

void main() {
  group('extractPlaceholders', () {
    Set<String> extractPlaceholders(String text) {
      final pattern = RegExp(r'\{(\w+)\}');
      return pattern.allMatches(text).map((m) => m.group(1)!).toSet();
    }

    test('extracts single placeholder', () {
      expect(extractPlaceholders('Hello, {name}!'), equals({'name'}));
    });

    test('extracts multiple placeholders', () {
      expect(
        extractPlaceholders('{firstName} {lastName}'),
        equals({'firstName', 'lastName'}),
      );
    });

    test('returns empty set for no placeholders', () {
      expect(extractPlaceholders('Hello World'), isEmpty);
    });

    test('extracts count placeholder', () {
      expect(extractPlaceholders('{count} items'), equals({'count'}));
    });
  });

  group('loadLocaleTranslations', () {
    late Directory tempDir;

    setUp(() {
      tempDir = Directory.systemTemp.createTempSync('shard_i18n_test_');
    });

    tearDown(() {
      tempDir.deleteSync(recursive: true);
    });

    test('loads and merges JSON files from locale directory', () async {
      final localeDir = Directory(p.join(tempDir.path, 'en'));
      localeDir.createSync();

      File(p.join(localeDir.path, 'core.json')).writeAsStringSync(
        json.encode({'Hello': 'Hello', 'World': 'World'}),
      );
      File(p.join(localeDir.path, 'auth.json')).writeAsStringSync(
        json.encode({'Sign in': 'Sign in'}),
      );

      final merged = <String, dynamic>{};
      await for (final entity in localeDir.list()) {
        if (entity is File && entity.path.endsWith('.json')) {
          final content = await entity.readAsString();
          final Map<String, dynamic> data = json.decode(content);
          merged.addAll(data);
        }
      }

      expect(merged, hasLength(3));
      expect(merged['Hello'], equals('Hello'));
      expect(merged['Sign in'], equals('Sign in'));
    });

    test('returns empty map for non-existent directory', () async {
      final localeDir = Directory(p.join(tempDir.path, 'nonexistent'));
      expect(localeDir.existsSync(), isFalse);
    });

    test('skips invalid JSON files gracefully', () async {
      final localeDir = Directory(p.join(tempDir.path, 'en'));
      localeDir.createSync();

      File(p.join(localeDir.path, 'valid.json')).writeAsStringSync(
        json.encode({'key': 'value'}),
      );
      File(p.join(localeDir.path, 'invalid.json')).writeAsStringSync(
        'not valid json {{{',
      );

      final merged = <String, dynamic>{};
      await for (final entity in localeDir.list()) {
        if (entity is File && entity.path.endsWith('.json')) {
          try {
            final content = await entity.readAsString();
            final Map<String, dynamic> data = json.decode(content);
            merged.addAll(data);
          } catch (_) {
            // Skip invalid files
          }
        }
      }

      expect(merged, hasLength(1));
      expect(merged['key'], equals('value'));
    });
  });

  group('checkPlaceholderParity', () {
    Set<String> extractPlaceholders(String text) {
      final pattern = RegExp(r'\{(\w+)\}');
      return pattern.allMatches(text).map((m) => m.group(1)!).toSet();
    }

    List<String> checkPlaceholderParity(
      Map<String, dynamic> reference,
      Map<String, dynamic> target,
    ) {
      final errors = <String>[];
      for (final key in reference.keys) {
        if (!target.containsKey(key)) continue;
        final refValue = reference[key];
        final targetValue = target[key];
        if (refValue is String && targetValue is String) {
          final refPlaceholders = extractPlaceholders(refValue);
          final targetPlaceholders = extractPlaceholders(targetValue);
          if (!refPlaceholders.every((p) => targetPlaceholders.contains(p))) {
            errors.add('$key: placeholders mismatch');
          }
        }
      }
      return errors;
    }

    test('returns empty for matching placeholders', () {
      final ref = {'Hello, {name}!': 'Hello, {name}!'};
      final target = {'Hello, {name}!': 'Hallo, {name}!'};
      expect(checkPlaceholderParity(ref, target), isEmpty);
    });

    test('detects missing placeholder in target', () {
      final ref = {'Hello, {name}!': 'Hello, {name}!'};
      final target = {'Hello, {name}!': 'Hallo!'};
      expect(checkPlaceholderParity(ref, target), hasLength(1));
    });

    test('skips keys not in target', () {
      final ref = {'Hello, {name}!': 'Hello, {name}!'};
      final target = <String, dynamic>{};
      expect(checkPlaceholderParity(ref, target), isEmpty);
    });
  });
